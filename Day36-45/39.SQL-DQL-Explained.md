# SQL DQL Explained

Now we'll explore query operations in DML using the school course selection database created previously. Whether for developers or data analysts, queries are crucial for retrieving needed data from relational databases. Re-execute the DDL from the database creation lesson and the DML from the data insertion lesson to ensure tables and data are correct before proceeding.

```sql
USE school;

-- Query all student information
SELECT stu_id,
       stu_name,
       stu_sex,
       stu_birth,
       stu_addr,
       col_id
  FROM tb_student;

-- Query student ID, name, and address (projection and aliases)
SELECT stu_id AS ID,
       stu_name AS Name,
       stu_addr AS Address
  FROM tb_student;

-- Query all course names and credits (projection and aliases)
SELECT cou_name AS Course,
       cou_credit AS Credits
  FROM tb_course;

-- Query female students' names and birth dates (filtering)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0;

-- Query female students from Chengdu (filtering with AND)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       AND stu_addr = 'Chengdu, Sichuan';

-- Query students from Chengdu or female students (filtering with OR)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 0
       OR stu_addr = 'Chengdu, Sichuan';

-- Query all students born in the 1980s (filtering)
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE '1980-1-1' <= stu_birth 
       AND stu_birth <= '1989-12-31';
       
SELECT stu_name,
       stu_sex,
       stu_birth
  FROM tb_student
 WHERE stu_birth BETWEEN '1980-1-1' AND '1989-12-31';

-- Query courses with credits greater than 2 (filtering)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit > 2;

-- Query courses with odd-numbered credits (filtering)
SELECT cou_name,
	   cou_credit
  FROM tb_course
 WHERE cou_credit MOD 2 <> 0;

-- Query students with scores above 90 in course 1111 (filtering)
SELECT stu_id
  FROM tb_record
 WHERE cou_id = 1111
       AND score > 90;

-- Query student named "Yang Guo" with name and gender (filtering and conditional)
SELECT stu_name AS Name, 
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name = 'Yang Guo';
 
SELECT stu_name AS Name, 
       IF(stu_sex, 'Male', 'Female') AS Gender
  FROM tb_student
 WHERE stu_name = 'Yang Guo';
    
-- Query students with surname "Yang" (pattern matching with LIKE)
-- Wildcard % matches zero or any number of characters
SELECT stu_name AS Name, 
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang%';

-- Query students with surname "Yang" and 2-character names (pattern matching with _)
-- Wildcard _ matches exactly one character
SELECT stu_name AS Name, 
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang_';

-- Query students with surname "Yang" and 3-character names (pattern matching)
SELECT stu_name AS Name, 
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender
  FROM tb_student
 WHERE stu_name LIKE 'Yang__';
 
-- Query students whose ID ends with 3 (pattern matching)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_id LIKE '%3';

-- Query students with "Bu" or "Yan" in their name (pattern matching and union)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Bu%'
       OR stu_name LIKE '%Yan%';
       
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Bu%'
 UNION
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name LIKE '%Yan%';

-- Query students with surname "Yang" or "Lin" and 3-character names (regex matching)
SELECT stu_id,
       stu_name
  FROM tb_student
 WHERE stu_name REGEXP '[Lin Yang][A-Z ]{1,}';

-- Query students without address (NULL handling)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) = ''
       OR stu_addr is null;
 
-- Query students with address (NULL handling)
SELECT stu_name
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Query all course selection dates (distinct values)
SELECT DISTINCT sel_date
  FROM tb_record;

-- Query student addresses (distinct values)
SELECT DISTINCT stu_addr
  FROM tb_student
 WHERE TRIM(stu_addr) <> ''
       AND stu_addr is not null;

-- Query male students' names and birth dates, ordered by age descending (sorting)
SELECT stu_name,
       stu_birth
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY stu_birth ASC;
 
-- Supplement: Convert birth dates to age above (date and numeric functions)
SELECT stu_name AS Name,
       FLOOR(DATEDIFF(CURDATE(), stu_birth) / 365) AS Age
  FROM tb_student
 WHERE stu_sex = 1
 ORDER BY Age DESC;

-- Query oldest student's birth date (aggregate function)
SELECT MIN(stu_birth)
  FROM tb_student;

-- Query youngest student's birth date (aggregate function)
SELECT MAX(stu_birth)
  FROM tb_student;

-- Query course 1111's highest exam score (aggregate function)
SELECT MAX(score)
  FROM tb_record
 WHERE cou_id = 1111;

-- Query student 1001's minimum, maximum, average score, standard deviation, variance (aggregate functions)
SELECT MIN(score) AS Min_Score,
       MAX(score) AS Max_Score,
	   ROUND(AVG(score), 1) AS Avg_Score,
       STDDEV(score) AS Std_Dev,
       VARIANCE(score) AS Variance
  FROM tb_record
 WHERE stu_id = 1001;

-- Query student 1001's average score, counting NULL as 0 (aggregate functions)
SELECT ROUND(SUM(score) / COUNT(*), 1) AS Avg_Score
  FROM tb_record
 WHERE stu_id = 1001;

-- Query number of male and female students (grouping and aggregate functions)
SELECT CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender,
       COUNT(*) AS Count
  FROM tb_student
 GROUP BY stu_sex;

-- Query student count per college (grouping and aggregate functions)
SELECT col_id AS College_ID,
       COUNT(*) AS Count
  FROM tb_student
 GROUP BY col_id
  WITH ROLLUP;

-- Query male and female student count per college (grouping and aggregate functions)
SELECT col_id AS College_ID,
       CASE stu_sex WHEN 1 THEN 'Male' ELSE 'Female' END AS Gender,
       COUNT(*) AS Count
  FROM tb_student
 GROUP BY col_id, stu_sex;

-- Query each student's ID and average score (grouping and aggregate functions)
SELECT stu_id AS ID,
	   ROUND(AVG(score), 1) AS Avg_Score
  FROM tb_record
 GROUP BY stu_id;

-- Query students with average score >= 90 (filtering after grouping)
SELECT stu_id AS ID,
	   ROUND(AVG(score), 1) AS Avg_Score
  FROM tb_record
 GROUP BY stu_id
HAVING Avg_Score >= 90;

-- Query average score >= 90 for courses 1111, 2222, 3333 (filtering before and after grouping)
SELECT stu_id AS ID,
	   ROUND(AVG(score), 1) AS Avg_Score
  FROM tb_record
 WHERE cou_id in (1111, 2222, 3333)
 GROUP BY stu_id
HAVING Avg_Score >= 90
 ORDER BY Avg_Score ASC;

-- Query oldest student's name (subquery)
SELECT stu_name
  FROM tb_student
 WHERE stu_birth = (SELECT MIN(stu_birth)
                      FROM tb_student);

-- Query students taking more than two courses (subquery and set operations)
SELECT stu_name
  FROM tb_student
 WHERE stu_id in (SELECT stu_id
				    FROM tb_record
				   GROUP BY stu_id
				  HAVING COUNT(*) > 2);

-- Query student name, birth date, and college name (table joins)
SELECT stu_name,
       stu_birth,
       col_name
  FROM tb_student AS t1, tb_college AS t2
 WHERE t1.col_id = t2.col_id;
 
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student INNER JOIN tb_college
       ON tb_student.col_id = tb_college.col_id;

SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student NATURAL JOIN tb_college;
  
SELECT stu_name,
       stu_birth,
	   col_name
  FROM tb_student CROSS JOIN tb_college;

-- Query student name, course name, and score (table joins)
SELECT stu_name,
       cou_name,
	   score
  FROM tb_student, tb_course, tb_record
 WHERE tb_student.stu_id = tb_record.stu_id
       AND tb_course.cou_id = tb_record.cou_id
       AND score is not null;

SELECT stu_name,
       cou_name,
	   score
  FROM tb_student 
       INNER JOIN tb_record
	       ON tb_student.stu_id = tb_record.stu_id
	   INNER JOIN tb_course
	       ON tb_course.cou_id = tb_record.cou_id
 WHERE score is not null;
 
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null;

-- Supplement: Get first 5 records from above (pagination)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5;

-- Supplement: Get records 6-10 (pagination)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 5
OFFSET 5;

-- Supplement: Get records 11-15 (pagination)
SELECT stu_name,
       cou_name,
       score
  FROM tb_student 
	   NATURAL JOIN tb_record
       NATURAL JOIN tb_course
 WHERE score is not null
 ORDER BY cou_id ASC, score DESC
 LIMIT 10, 5;

-- Query course students' names and average scores (subquery and table joins)
-- Error Code: 1248. Every derived table must have its own alias
SELECT stu_name,
	   avg_score
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            ROUND(AVG(score), 1) AS avg_score
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Query student name and course count (subquery and table joins)
SELECT stu_name,
	   total
  FROM tb_student
       NATURAL JOIN (SELECT stu_id,
                            COUNT(*) AS total
                       FROM tb_record
                      GROUP BY stu_id) as tmp;

-- Query each student's name and course count (subquery and left outer join)
SELECT stu_name AS Name,
	   COALESCE(total, 0) AS Course_Count
  FROM tb_student AS t1
	   LEFT JOIN (SELECT stu_id,
			             COUNT(*) AS total
		            FROM tb_record
				   GROUP BY stu_id) AS t2
	       ON t1.stu_id = t2.stu_id;
```

Several important clarifications:

1. MySQL currently doesn't support full outer joins. Above, we achieve full outer joins using `union` to combine left and right outer join results. The diagram below helps understand join operations.

2. MySQL supports multiple operator types: arithmetic operators (`+`, `-`, `*`, `/`, `%`), comparison operators (`=`, `<>`, `<=>`, `<`, `<=`, `>`, `>=`, `BETWEEN...AND..`, `IN`, `IS NULL`, `IS NOT NULL`, `LIKE`, `RLIKE`, `REGEXP`), logical operators (`NOT`, `AND`, `OR`, `XOR`), and bitwise operators (`&`, `|`, `^`, `~`, `>>`, `<<`). You can use these operators in DML to process data.

3. When querying, use functions in `SELECT` statements and clauses like `WHERE`, `ORDER BY`, `HAVING`. These include string functions, numeric functions, date-time functions, and control flow functions, as shown in the tables below.

   **Common string functions:**

   | Function                        | Purpose                                                 |
   | ------------------------------- | ------------------------------------------------------- |
   | `CONCAT`                        | Concatenate multiple strings into one                   |
   | `FORMAT`                        | Format numeric value to string with specified decimals  |
   | `FROM_BASE64` / `TO_BASE64`     | BASE64 decode/encode                                    |
   | `BIN` / `OCT` / `HEX`           | Convert numeric value to binary/octal/hexadecimal      |
   | `LOCATE`                        | Find substring position in string                       |
   | `LEFT` / `RIGHT`                | Return left/right portion of string with specified length |
   | `LENGTH` / `CHAR_LENGTH`        | Return string length in bytes/characters                |
   | `LOWER` / `UPPER`               | Return lowercase/uppercase string                       |
   | `LPAD` / `RPAD`                 | Pad string left/right with specified characters if needed |
   | `LTRIM` / `RTRIM`               | Remove leading/trailing spaces                          |
   | `ORD` / `CHAR`                  | Return character encoding/character for encoding        |
   | `STRCMP`                        | Compare strings; return -1/0/1 for less/equal/greater  |
   | `SUBSTRING`                     | Return substring in specified range                     |

   **Common numeric functions:**

   | Function                                                 | Purpose                            |
   | -------------------------------------------------------- | ---------------------------------- |
   | `ABS`                                                    | Return absolute value              |
   | `CEILING` / `FLOOR`                                      | Return ceiling/floor value         |
   | `CONV`                                                   | Convert number between bases       |
   | `CRC32`                                                  | Calculate cyclic redundancy check  |
   | `EXP` / `LOG` / `LOG2` / `LOG10`                         | Calculate exponential/logarithms  |
   | `POW`                                                    | Calculate power                    |
   | `RAND`                                                   | Return random number in [0,1)     |
   | `ROUND`                                                  | Round number                       |
   | `SQRT`                                                   | Calculate square root              |
   | `TRUNCATE`                                               | Truncate number to precision       |
   | `SIN` / `COS` / `TAN` / `COT` / `ASIN` / `ACOS` / `ATAN` | Trigonometric functions           |

   **Common date-time functions:**

   | Function                                      | Purpose                                   |
   | --------------------------------------------- | ----------------------------------------- |
   | `CURDATE` / `CURTIME` / `NOW`                 | Get current date/time/date-time          |
   | `ADDDATE` / `SUBDATE`                         | Add/subtract dates and return result      |
   | `DATE` / `TIME`                               | Extract date/time from string             |
   | `YEAR` / `MONTH` / `DAY`                      | Extract year/month/day from date          |
   | `HOUR` / `MINUTE` / `SECOND`                  | Extract hour/minute/second from time      |
   | `DATEDIFF` / `TIMEDIFF` / `TIMESTAMPDIFF`     | Return difference in days/hours/time units |
   | `MAKEDATE` / `MAKETIME`                       | Create date/time                          |

   **Common control flow functions:**

   | Function     | Purpose                                          |
   | ------------ | ------------------------------------------------ |
   | `IF`         | Return different values based on condition       |
   | `IFNULL`     | Return specified value if NULL, else original    |
   | `NULLIF`     | Return NULL if equal, else first expression      |

   **Other common functions:**

   | Function                       | Purpose                           |
   | ------------------------------ | --------------------------------- |
   | `MD5` / `SHA1` / `SHA2`        | Return string hash digest         |
   | `CHARSET` / `COLLATION`        | Return character set/collation    |
   | `USER` / `CURRENT_USER`        | Return current user               |
   | `DATABASE`                     | Return current database name      |
   | `VERSION`                      | Return database version           |
   | `FOUND_ROWS` / `ROW_COUNT`     | Return query/affected rows count  |
   | `LAST_INSERT_ID`               | Return last auto-increment value  |
   | `UUID` / `UUID_SHORT`          | Return globally unique identifier |
